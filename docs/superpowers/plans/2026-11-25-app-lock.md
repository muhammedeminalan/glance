```swift title="tools/app_lock_selftest.swift"
//
//  app_lock_selftest.swift
//  glance (tools)
//
//  Offline test for AppLockState's relock policies. This project has no test target, so this compiles as a
//  standalone script:
//
//      swiftc -parse-as-library -o /tmp/app_lock_selftest \
//        glance/AppLock/AppLockState.swift tools/app_lock_selftest.swift && /tmp/app_lock_selftest
//

import Foundation

@main
enum AppLockSelfTest {
    static var failures = 0

    static func check(_ condition: Bool, _ message: String) {
        if condition { print("ok   \(message)") } else { failures += 1; print("FAIL \(message)") }
    }

    static func main() {
        let t0 = Date(timeIntervalSince1970: 1_800_000_000)
        let app = "net.whatsapp.WhatsApp"
        var state = AppLockState()

        for policy in AppLockRelockPolicy.allCases {
            check(state.needsVerification(bundleID: app, pid: 10, policy: policy, now: t0),
                  "\(policy): an app never unlocked needs a check")
        }

        state.markUnlocked(bundleID: app, pid: 10, at: t0)
        for policy in AppLockRelockPolicy.allCases {
            check(!state.needsVerification(bundleID: app, pid: 10, policy: policy, now: t0.addingTimeInterval(1)),
                  "\(policy): coming back within the 2 s grace needs no check")
        }
        check(state.needsVerification(bundleID: app, pid: 11, policy: .onQuit, now: t0.addingTimeInterval(1)),
              "a relaunched app (new pid) needs a check")
        check(!state.needsVerification(bundleID: app, pid: 10, policy: .onQuit, now: t0.addingTimeInterval(3600)),
              "onQuit: still open an hour later")
        check(state.needsVerification(bundleID: app, pid: 10, policy: .everyActivation, now: t0.addingTimeInterval(10)),
              "everyActivation: checks again after the grace")

        state.markDeactivated(bundleID: app, pid: 10, at: t0.addingTimeInterval(60))
        check(!state.needsVerification(bundleID: app, pid: 10, policy: .afterFiveMinutesAway, now: t0.addingTimeInterval(60 + 299)),
              "afterFiveMinutesAway: back within 5 min needs no check")
        check(state.needsVerification(bundleID: app, pid: 10, policy: .afterFiveMinutesAway, now: t0.addingTimeInterval(60 + 301)),
              "afterFiveMinutesAway: away over 5 min needs a check")
        state.markActivated(bundleID: app, pid: 10)
        check(!state.needsVerification(bundleID: app, pid: 10, policy: .afterFiveMinutesAway, now: t0.addingTimeInterval(60 + 600)),
              "afterFiveMinutesAway: coming back resets the away timer")
        state.markDeactivated(bundleID: app, pid: 10, at: t0.addingTimeInterval(1000))
        state.markDeactivated(bundleID: app, pid: 10, at: t0.addingTimeInterval(1200))
        check(state.needsVerification(bundleID: app, pid: 10, policy: .afterFiveMinutesAway, now: t0.addingTimeInterval(1000 + 301)),
              "afterFiveMinutesAway: the first deactivation starts the timer")

        state.markUnlocked(bundleID: app, pid: 10, at: t0)
        state.markTerminated(bundleID: app)
        check(state.needsVerification(bundleID: app, pid: 10, policy: .onQuit, now: t0.addingTimeInterval(10)),
              "quitting the app locks it again")

        state.markUnlocked(bundleID: app, pid: 10, at: t0)
        state.markUnlocked(bundleID: "com.apple.Notes", pid: 20, at: t0)
        state.lockAll()
        check(state.needsVerification(bundleID: app, pid: 10, policy: .onQuit, now: t0.addingTimeInterval(10))
              && state.needsVerification(bundleID: "com.apple.Notes", pid: 20, policy: .onQuit, now: t0.addingTimeInterval(10)),
              "lockAll locks every app")

        print(failures == 0 ? "ALL PASSED" : "\(failures) FAILED")
        exit(failures == 0 ? 0 : 1)
    }
}
```

```swift title="glance/AppLock/AppLockState.swift"
//
//  AppLockState.swift
//  glance
//
//  Which locked apps the user has already opened, and whether bringing one to the front again needs a fresh check.
//  Pure value logic with no AppKit, so tools/app_lock_selftest.swift can exercise it offline.
//

import Foundation

/// When an app the user opened locks again. The Mac locking or sleeping relocks every app regardless.
nonisolated enum AppLockRelockPolicy: String, CaseIterable, Identifiable, Sendable {
    case onQuit
    case afterFiveMinutesAway
    case everyActivation

    var id: String { rawValue }
}

nonisolated struct AppLockState: Sendable {
    /// `afterFiveMinutesAway`'s limit.
    static let awayLimit: TimeInterval = 5 * 60
    /// Activations this soon after an unlock are the app returning to the front because of that unlock, not the user coming back.
    static let reactivationGrace: TimeInterval = 2

    private struct Unlock: Sendable {
        var pid: Int32
        var unlockedAt: Date
        var deactivatedAt: Date?
    }

    private var unlocks: [String: Unlock] = [:]

    init() {}

    func needsVerification(bundleID: String, pid: Int32, policy: AppLockRelockPolicy, now: Date) -> Bool {
        guard let unlock = unlocks[bundleID], unlock.pid == pid else { return true }
        if now.timeIntervalSince(unlock.unlockedAt) < Self.reactivationGrace { return false }
        switch policy {
        case .onQuit:
            return false
        case .afterFiveMinutesAway:
            guard let away = unlock.deactivatedAt else { return false }
            return now.timeIntervalSince(away) > Self.awayLimit
        case .everyActivation:
            return true
        }
    }

    mutating func markUnlocked(bundleID: String, pid: Int32, at now: Date) {
        unlocks[bundleID] = Unlock(pid: pid, unlockedAt: now, deactivatedAt: nil)
    }

    /// The app came back without needing a check, so its away timer starts over.
    mutating func markActivated(bundleID: String, pid: Int32) {
        guard unlocks[bundleID]?.pid == pid else { return }
        unlocks[bundleID]?.deactivatedAt = nil
    }

    /// Only the first deactivation counts; a later one must not push the away timer back.
    mutating func markDeactivated(bundleID: String, pid: Int32, at now: Date) {
        guard unlocks[bundleID]?.pid == pid, unlocks[bundleID]?.deactivatedAt == nil else { return }
        unlocks[bundleID]?.deactivatedAt = now
    }

    mutating func markTerminated(bundleID: String) {
        unlocks[bundleID] = nil
    }

    mutating func lockAll() {
        unlocks.removeAll()
    }
}
```

```swift title="glance/AppLock/FaceVerifier.swift"
//
//  FaceVerifier.swift
//  glance
//
//  A short "is this the enrolled user?" check for App Lock: recognition plus liveness, kept apart from the lock-screen
//  scan so that flow stays untouched.
//

import Foundation
import CoreGraphics

@MainActor
final class FaceVerifier {
    enum Outcome: Equatable {
        case matched
        case notRecognized
        /// Nothing to compare against (session locked, nobody enrolled) or the camera couldn't start.
        case unavailable
    }

    private let camera: CameraManager
    private let pipeline: FaceRecognitionPipeline

    init(camera: CameraManager, pipeline: FaceRecognitionPipeline) {
        self.camera = camera
        self.pipeline = pipeline
    }

    func verify(timeout: TimeInterval = 5) async -> Outcome {
        guard SecureCredentialManager.isSessionUnlocked else { return .unavailable }
        let store = FaceEnrollmentStore.shared
        store.reloadIfUnlocked()
        let identities = store.activeIdentities
        guard !identities.isEmpty else { return .unavailable }

        await camera.start()
        defer { camera.stop() }
        guard camera.errorMessage == nil else { return .unavailable }

        let settings = GlanceSettings.shared
        let threshold = settings.matchThreshold
        let livenessEnabled = settings.livenessChecksEnabled
        let liveness = LivenessAnalyzer()
        liveness.modeProvider = { GlanceSettings.shared.livenessMode }
        var livenessConfirmed = !livenessEnabled
        var lastFrameID: UInt64?
        var lastFaceBox: CGRect?
        let deadline = Date().addingTimeInterval(timeout)

        while Date() < deadline, !Task.isCancelled {
            guard let frame = camera.currentFrame, frame.id != lastFrameID else {
                try? await Task.sleep(nanoseconds: 20_000_000)
                continue
            }
            lastFrameID = frame.id

            let pipeline = self.pipeline
            let previousBox = lastFaceBox
            let analysis = await Task.detached(priority: .userInitiated) { () -> (FaceRecognitionResult, LivenessFrame)? in
                guard let result = try? pipeline.recognize(in: frame.image, preferNear: previousBox) else { return nil }
                let faceCrop = CameraManager.renderCrop(from: frame, imageRect: result.face.boundingBox)
                return (result, LivenessFeatureExtractor.extract(from: result, frame: frame.image, faceCrop: faceCrop))
            }.value

            guard let (result, livenessFrame) = analysis else {
                lastFaceBox = nil
                try? await Task.sleep(nanoseconds: 20_000_000)
                continue
            }
            lastFaceBox = result.face.normalizedBoundingBox

            if livenessEnabled {
                switch liveness.observe(livenessFrame).decision {
                case .denied: return .notRecognized
                case .confirmed: livenessConfirmed = true
                case .pending: break
                }
            }

            let scored = pipeline.score(result.embedding, against: identities)
            if pipeline.bestMatch(in: scored, threshold: threshold) != nil, livenessConfirmed {
                return .matched
            }
        }
        return .notRecognized
    }
}
```

```swift title="glance/AppLock/AppLockOverlay.swift"
//
//  AppLockOverlay.swift
//  glance
//
//  The full-screen blur covering a locked app while its owner proves who they are.
//

import SwiftUI
import AppKit

@Observable
@MainActor
final class AppLockOverlayModel {
    enum Phase: Equatable {
        case scanning
        case notRecognized
        case authenticating
    }

    var appName = ""
    var appIcon: NSImage?
    var phase: Phase = .scanning
    var onTouchID: () -> Void = {}
    var onRetry: () -> Void = {}
    var onCancel: () -> Void = {}
}

/// Borderless but key-capable, so the buttons work and keystrokes can't reach the locked app underneath. Never main, so
/// `AppDelegate`'s Dock-icon bookkeeping (which only counts main-capable windows) ignores it.
private final class AppLockOverlayWindow: NSWindow {
    override var canBecomeKey: Bool { true }
    override var canBecomeMain: Bool { false }
}

@MainActor
final class AppLockOverlayController {
    let model = AppLockOverlayModel()
    private var windows: [NSWindow] = []

    /// One window per screen so the locked app can't show through on a second display; only the main screen gets controls.
    func present(appName: String, appIcon: NSImage?) {
        model.appName = appName
        model.appIcon = appIcon
        model.phase = .scanning
        guard windows.isEmpty else { return }

        let controlsScreen = NSScreen.main ?? NSScreen.screens.first
        var keyWindow: NSWindow?
        for screen in NSScreen.screens {
            let showsControls = screen == controlsScreen
            let window = AppLockOverlayWindow(contentRect: screen.frame, styleMask: [.borderless], backing: .buffered, defer: false)
            // Floating, not screen-saver: the Touch ID / password sheet has to be able to appear above it.
            window.level = .floating
            window.collectionBehavior = [.canJoinAllSpaces, .fullScreenAuxiliary, .stationary, .ignoresCycle]
            window.isOpaque = false
            window.backgroundColor = .clear
            window.hasShadow = false
            window.isReleasedWhenClosed = false
            window.contentView = NSHostingView(rootView: AppLockOverlayView(model: model, showsControls: showsControls))
            window.setFrame(screen.frame, display: true)
            window.orderFrontRegardless()
            windows.append(window)
            if showsControls { keyWindow = window }
        }
        NSApp.activate()
        (keyWindow ?? windows.first)?.makeKey()
    }

    func dismiss() {
        for window in windows {
            window.orderOut(nil)
            window.close()
        }
        windows.removeAll()
    }
}

struct AppLockOverlayView: View {
    @Bindable var model: AppLockOverlayModel
    let showsControls: Bool

    var body: some View {
        ZStack {
            VisualEffectView(material: .fullScreenUI, blendingMode: .behindWindow)
            Color.black.opacity(0.35)
            if showsControls {
                controls
            }
        }
        .ignoresSafeArea()
    }

    private var controls: some View {
        VStack(spacing: 18) {
            if let icon = model.appIcon {
                Image(nsImage: icon)
                    .resizable()
                    .frame(width: 84, height: 84)
            }
            Text(String(localized: "\(model.appName) is locked"))
                .font(.system(size: 26, weight: .bold))
                .foregroundStyle(.white)
            Image(systemName: "faceid")
                .font(.system(size: 40, weight: .regular))
                .foregroundStyle(.white.opacity(0.9))
                .symbolEffect(.pulse, isActive: model.phase == .scanning)
            Text(statusText)
                .font(.system(size: 14))
                .foregroundStyle(.white.opacity(0.75))
            HStack(spacing: 12) {
                PillButton(title: String(localized: "Cancel"), style: .secondary) { model.onCancel() }
                if model.phase == .notRecognized {
                    PillButton(title: String(localized: "Try again"), style: .secondary) { model.onRetry() }
                }
                PillButton(title: String(localized: "Open with Touch ID"), isEnabled: model.phase != .authenticating) {
                    model.onTouchID()
                }
            }
            .padding(.top, 8)
        }
        .padding(40)
        .onExitCommand { model.onCancel() }
    }

    private var statusText: String {
        switch model.phase {
        case .scanning: return String(localized: "Look at the camera to open it")
        case .notRecognized: return String(localized: "Face not recognized")
        case .authenticating: return String(localized: "Authenticating…")
        }
    }
}
```

```swift title="glance/AppLock/AppLockMonitor.swift"
//
//  AppLockMonitor.swift
//  glance
//
//  Watches which app comes to the front and covers locked ones until the user's face, Touch ID or password opens them.
//

import AppKit
import LocalAuthentication

@MainActor
final class AppLockMonitor: NSObject {
    private var state = AppLockState()
    private let verifier: FaceVerifier
    private let overlay = AppLockOverlayController()
    /// The locked app currently behind the overlay, if any.
    private var pendingApp: NSRunningApplication?
    private var checkTask: Task<Void, Never>?

    init(pipeline: FaceRecognitionPipeline) {
        // Its own camera, not the lock screen's: a late `stop()` on a shared camera can kill a lock-screen scan mid-warm-up.
        verifier = FaceVerifier(camera: CameraManager(), pipeline: pipeline)
        super.init()

        overlay.model.onTouchID = { [weak self] in self?.authenticateWithDeviceOwner() }
        overlay.model.onRetry = { [weak self] in self?.startFaceCheck() }
        overlay.model.onCancel = { [weak self] in self?.cancel(hidingApp: true) }

        let workspace = NSWorkspace.shared.notificationCenter
        workspace.addObserver(self, selector: #selector(appDidActivate(_:)), name: NSWorkspace.didActivateApplicationNotification, object: nil)
        workspace.addObserver(self, selector: #selector(appDidDeactivate(_:)), name: NSWorkspace.didDeactivateApplicationNotification, object: nil)
        workspace.addObserver(self, selector: #selector(appDidTerminate(_:)), name: NSWorkspace.didTerminateApplicationNotification, object: nil)
        workspace.addObserver(self, selector: #selector(lockEverything(_:)), name: NSWorkspace.willSleepNotification, object: nil)
        DistributedNotificationCenter.default().addObserver(
            self, selector: #selector(lockEverything(_:)),
            name: Notification.Name("com.apple.screenIsLocked"), object: nil
        )

        // An app that is already in front when Glance launches never posts an activation.
        DispatchQueue.main.async { [weak self] in
            self?.handleActivation(of: NSWorkspace.shared.frontmostApplication)
        }
    }

    private static func app(from notification: Notification) -> NSRunningApplication? {
        notification.userInfo?[NSWorkspace.applicationUserInfoKey] as? NSRunningApplication
    }

    @objc private func appDidActivate(_ notification: Notification) {
        handleActivation(of: Self.app(from: notification))
    }

    @objc private func appDidDeactivate(_ notification: Notification) {
        guard let app = Self.app(from: notification), let bundleID = app.bundleIdentifier else { return }
        state.markDeactivated(bundleID: bundleID, pid: app.processIdentifier, at: Date())
    }

    @objc private func appDidTerminate(_ notification: Notification) {
        guard let app = Self.app(from: notification), let bundleID = app.bundleIdentifier else { return }
        state.markTerminated(bundleID: bundleID)
        if pendingApp?.processIdentifier == app.processIdentifier {
            cancel(hidingApp: false)
        }
    }

    @objc private func lockEverything(_ notification: Notification) {
        state.lockAll()
        if pendingApp != nil {
            cancel(hidingApp: true)
        }
    }

    private func handleActivation(of app: NSRunningApplication?) {
        guard let app, let bundleID = app.bundleIdentifier, bundleID != Bundle.main.bundleIdentifier else { return }
        if let pendingApp {
            guard pendingApp.processIdentifier != app.processIdentifier else { return }
            // The user moved on mid-check: keep the locked app out of sight.
            cancel(hidingApp: true)
        }
        let settings = GlanceSettings.shared
        guard settings.lockedAppBundleIDs.contains(bundleID) else { return }
        if state.needsVerification(bundleID: bundleID, pid: app.processIdentifier, policy: settings.appLockRelockPolicy, now: Date()) {
            begin(app)
        } else {
            state.markActivated(bundleID: bundleID, pid: app.processIdentifier)
        }
    }

    private func begin(_ app: NSRunningApplication) {
        pendingApp = app
        overlay.present(appName: app.localizedName ?? app.bundleIdentifier ?? "", appIcon: app.icon)
        startFaceCheck()
    }

    private func startFaceCheck() {
        checkTask?.cancel()
        overlay.model.phase = .scanning
        checkTask = Task { [weak self] in
            guard let self else { return }
            let outcome = await self.verifier.verify()
            guard !Task.isCancelled, self.pendingApp != nil else { return }
            switch outcome {
            case .matched:
                self.unlockPendingApp()
            case .notRecognized:
                self.overlay.model.phase = .notRecognized
            case .unavailable:
                self.authenticateWithDeviceOwner()
            }
        }
    }

    private func authenticateWithDeviceOwner() {
        guard let app = pendingApp else { return }
        checkTask?.cancel()
        overlay.model.phase = .authenticating
        let name = app.localizedName ?? ""
        checkTask = Task { [weak self] in
            let context = LAContext()
            let approved = (try? await context.evaluatePolicy(
                .deviceOwnerAuthentication,
                localizedReason: String(localized: "Authenticate to open \(name)")
            )) ?? false
            guard let self, !Task.isCancelled, self.pendingApp?.processIdentifier == app.processIdentifier else { return }
            if approved {
                self.unlockPendingApp()
            } else {
                self.overlay.model.phase = .notRecognized
            }
        }
    }

    private func unlockPendingApp() {
        guard let app = pendingApp, let bundleID = app.bundleIdentifier else { return }
        checkTask?.cancel()
        checkTask = nil
        state.markUnlocked(bundleID: bundleID, pid: app.processIdentifier, at: Date())
        pendingApp = nil
        overlay.dismiss()
        _ = app.activate()
    }

    private func cancel(hidingApp: Bool) {
        checkTask?.cancel()
        checkTask = nil
        let app = pendingApp
        pendingApp = nil
        overlay.dismiss()
        if hidingApp {
            _ = app?.hide()
        }
    }
}
```

```swift title="glance/Settings/Pages/AppLockSettingsPage.swift"
//
//  AppLockSettingsPage.swift
//  glance
//
//  Pick which apps need your face before they show their windows, and when they lock again.
//

import SwiftUI
import AppKit
import UniformTypeIdentifiers

struct AppLockSettingsPage: View {
    @Bindable private var settings = GlanceSettings.shared

    var body: some View {
        VStack(alignment: .leading, spacing: SettingsMetrics.rowSpacing) {
            SettingsCaption(text: String(localized: "A locked app stays covered until Glance recognizes your face, or you use Touch ID or your password. It keeps casual onlookers out; it doesn't replace a separate macOS user account."))
                .padding(.horizontal, SettingsMetrics.sectionTitleHorizontalInset)

            SettingsGroup {
                SettingsRowContent(title: String(localized: "Lock again")) {
                    SettingsMenuPickerPill(label: Self.title(for: settings.appLockRelockPolicy)) {
                        ForEach(AppLockRelockPolicy.allCases) { policy in
                            Button(Self.title(for: policy)) { settings.appLockRelockPolicy = policy }
                        }
                    }
                }
            }

            VStack(alignment: .leading, spacing: 8) {
                SettingsSectionTitle(text: String(localized: "Locked apps"))
                SettingsGroup {
                    ForEach(settings.lockedAppBundleIDs, id: \.self) { bundleID in
                        LockedAppRow(bundleID: bundleID) {
                            settings.lockedAppBundleIDs.removeAll { $0 == bundleID }
                        }
                        SettingsGroupDivider()
                    }
                    SettingsRowContent(
                        title: settings.lockedAppBundleIDs.isEmpty
                            ? String(localized: "No apps are locked yet.")
                            : String(localized: "Lock another app")
                    ) {
                        SettingsPrimaryButton(title: String(localized: "Add…"), compact: true, action: addApps)
                    }
                }
            }
        }
    }

    private func addApps() {
        let panel = NSOpenPanel()
        panel.allowedContentTypes = [.application]
        panel.allowsMultipleSelection = true
        panel.canChooseDirectories = false
        panel.directoryURL = URL(fileURLWithPath: "/Applications", isDirectory: true)
        panel.prompt = String(localized: "Lock")
        guard panel.runModal() == .OK else { return }
        let ownBundleID = Bundle.main.bundleIdentifier
        for url in panel.urls {
            guard let bundleID = Bundle(url: url)?.bundleIdentifier, bundleID != ownBundleID,
                  !settings.lockedAppBundleIDs.contains(bundleID) else { continue }
            settings.lockedAppBundleIDs.append(bundleID)
        }
    }

    private static func title(for policy: AppLockRelockPolicy) -> String {
        switch policy {
        case .onQuit: return String(localized: "When the app quits")
        case .afterFiveMinutesAway: return String(localized: "5 minutes after switching away")
        case .everyActivation: return String(localized: "Every time it opens")
        }
    }
}

/// Icon, name and a remove button for one locked app. Falls back to the bundle ID if the app is no longer installed.
private struct LockedAppRow: View {
    let bundleID: String
    let onRemove: () -> Void

    var body: some View {
        let url = NSWorkspace.shared.urlForApplication(withBundleIdentifier: bundleID)
        HStack(spacing: 10) {
            Image(nsImage: url.map { NSWorkspace.shared.icon(forFile: $0.path) } ?? NSImage(named: NSImage.applicationIconName) ?? NSImage())
                .resizable()
                .frame(width: 28, height: 28)
            Text(url.map { FileManager.default.displayName(atPath: $0.path) } ?? bundleID)
                .font(SettingsMetrics.rowFont)
                .foregroundStyle(SettingsMetrics.textPrimary)
            Spacer(minLength: 8)
            Button(action: onRemove) {
                Image(systemName: "trash")
                    .font(.system(size: 12))
                    .foregroundStyle(SettingsMetrics.textPrimary)
                    .frame(width: 28, height: 28)
                    .background(SettingsMetrics.neutralButtonFill)
                    .clipShape(Circle())
            }
            .buttonStyle(.plain)
            .help(String(localized: "Stop locking this app"))
        }
        .padding(.horizontal, SettingsMetrics.rowHorizontalInset)
        .frame(minHeight: SettingsMetrics.rowHeight)
    }
}
```
